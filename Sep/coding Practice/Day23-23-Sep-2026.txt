Ansible - Deploy lambda Functions using AWS CLI
- name: Deploy AWS Lambda Functions
  hosts: all
  tasks:
   - name: create lambda Functions
     command: aws lambda create-function --function-name max-effort-lambda --runtime python3.8 --role arn:aws:iam::23432414413:role/lambda-execution-role --handler lambda_function.lambda_handler --zip-file file://lambda.zip

Terraform - Serverless API gateway with lambda Integration

resource "aws_api_gateway_rest_api" "max-effort_api" {
  name = "max-api"
  description = "max API"
}

resource "aws_lambda_function" "max_lambda" {
  function_name = "max-lambda"
  runtime = "python3.8"
  role = aws_iam_role.lambda_exec_role.arn
  handler = "lambda_function.lambda_handler"
  filename = "lambda.zip"
}

resource "aws_api_gateway_integration" "lambda_integration" {
  rest_api_id = aws_api_gateway_rest_api.max_api.rest_api.id
  resource_id = aws_api_gateway_resource.max_resource.resource.id
  http_method = aws_api_gateway_method.max_method.http_method
  integration_http_method = "POST"
  type = "AWS_PROXY"
  uri = aws_lambda_function.max_lambda.invoke_arn
}

jenkins - deploy Serverless application with lambda
pipeline {
  agent any 
  stages {
    stage('Build') {
      steps {
        echo 'Building Lambda Function...'
        sh 'cip -r lambda.zip lambda/'
      }
    }
    stage('Deploy to AWS Lambda') {
      steps {
        script {
          sh 'aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip'
        }
      }
    }
  }
}

Docker 
Multistage Docker build
#stage1
FROM mave:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/pathnex/java-build
COPY . /opt/pathnex/java-build/
RUN mvn clean package

#Stage 2
FROM tomcat:9
WORKDIR /usr/local/tomcat/webapps
COPY --from=builder /opt/pathnex/java-build/target/*.wat /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh" , "run"]
