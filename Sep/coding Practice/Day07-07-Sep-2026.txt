## Ansible Task - Install Nginx on a server

---
- name: Install nginx on max-effort server
  hosts: all
  become: yes

  tasks:
   - name: install nginx
     yum:
      name: nginx
      state: present

## Kubernetes - Deployment with 2 replicas

apiVersion: v1
kind: Deployment
metadata: 
 name: max-effort-deployment 
spec:
 replicas: 2
 selector: 
    matchLabels:
        app: max-effort-app
  template:
    metadata:
        labels:
            app: max-effort-app 
    spec:
        containers:
        - name: app
          image: nginx
          ports:
           - containers: 80

# Parameterized Builds
## Jenins Pipeline - Parameterized

pipeline {
    agent any
    paramaters {
        string(name: 'Environment', defaultValue: 'dev', description: 'Deployment Environment')
    }
    environment {
        Institue_name = "Pathnex"
    }
    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Deploy') {
            steps {
                sh 'echo "Deploying to $Environment by $Institue_name"'
            }
        }
    }
}
