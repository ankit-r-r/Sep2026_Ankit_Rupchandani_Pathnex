 name: dev-deployment

  app: dev-app

 replicas: 2

  matchLables:

Jenkins - Deploy to Multi-Cluster env

pipeline {
    agent any
     stages{
        stage('Build') {
            steps {
                echo 'Building Docker Image...'
                docker.build('web-app')
            }
        }
        stage('Test') {
            steps {
                echo 'Running Tests...'
            }
        }
        stage('Deploy to k8s') {
            steps {
                script {
                    if(env.CLUSTER == 'ap-sout') {
                        sh 'kubectl --context=ap-south apply -f k8s/deployment.yaml'
                    }else {
                        sh 'kubectl --context=us-west apply -f k8s/deployment.yaml'
                    }
                }
            }
        }
     }
}

GitLab - CI/CD - Multi - Region Deployment with k8s
stages:
 - build
 - push
 - deploy 

build:
 stage: build
 script:
  - docker build -t web-app . 

push:
 stage: push
  - docker login -u "$CI_User" -p "$CI_Password"
  - docker push web-app

deploy:
 stage: deploy
 script:
  - |
    if [ "$CI_Commit_Ref_name" == "main" ];  then
      kubectl --context=ap-south apply -f k8s/prod-deployment.yaml
    else
      kubectl --context=us-west apply -f k8s/dev-deployment.yaml
