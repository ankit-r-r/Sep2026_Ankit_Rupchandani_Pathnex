Ansible - Auto-Healing Web Servers with Systemd

---
- name: Configure Auto-healing for Web Servers
  hosts: all
  become: yes
  tasks:
   - name: Ensure nginx service is running
     service:
      name:
      state: started
      enables: yes
   - name: Setup auto-restart for nginx service
     lineinfile:
      path: /etc/systemd/system/nginx.service.d/override.conf 
      line: "Restart=always"
      create: yes
   - name: Reload systemd
     command: systemctl daemon-Reload

K8s - self healing with liveness and readiness probes

apiVersion: apps/v1
kind: Deployment
metadata:
 name: dev-web-apps
spec:
 replicas: 3
 selector:
  matchLabels:
   labels:
    app: dev-web-app
 template:
  metadata:
   labels:
    app: dev-web-app
  spec:
   containers:
    - name: nginx
      image: nginx
      ports:
       - containerPort: 80
      livenessProbe:
       httpGet:
        path: /healthz
        port: 80
       initialDelaySeconds: 5
       periodSeconds: 10
      readinessProbe:
       httpGet:
        path: /healthz
        port: 80
      initialDelaySeconds: 5
      periodSeconds: 10


Jenkindfile - Rolling Deployments with K8s
pipeline {
    agent any
    environment {
        IMAGE_NAME = 'dev-web-app'
    }
    stages {
        stage('build') {
            steps {
                echo 'Building Docker Image..'
                docker.build("${env.IMAGE_NAME}")
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests..'
            }
        }
        stage('Deploy to k8s') {
            steps {
                script {
                    sh 'kubectl apply -f k8s/deployment.yaml
                    sh 'kubectl apply status deploymeny/dev-web-app'
                }
            }
        }
    }
}
